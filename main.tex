local asterisk_marker = "*"
local cross_marker = "†"
local block_number = 1
local indentation = false
local awaiting_line_after_as erisk = false

local function color_markers(line)
  line = line:gsub("%*", "\\textcolor{breviaryRed}{*}")
  line = line:gsub("†", "\\textcolor{breviaryRed}{†}")
  line = line:gsub("/", "\\textcolor{breviaryRed}{/}")
  return line
end

function reset_psalmody_state()
  block_number = 1
  indentation = false
  awaiting_line_after_asterisk = false
end

function process_breviary_line(line)
  local trimmed_line = line:gsub("%s+$", "")
  
  if trimmed_line == "" then
    block_number = block_number + 1
    indentation = false
    awaiting_line_after_asterisk = false
    return "\\par\\vspace{\\psalmodyblockskip}%"
  end

  if trimmed_line:match("^\\end{psalmody}") then
    return line
  end

  if trimmed_line:sub(1, 6) == "\\input" then
    return line
  end

  local indent_this_line = indentation

  if block_number == 2 then
    return "\\noindent\\makebox[\\linewidth][c]{\\textcolor{breviaryRed}{\\textbf{"
      .. trimmed_line .. "}}}\\\\"
  end

  if trimmed_line:match("^%d+%s+[Aa]nt%.") or trimmed_line:match("^[Aa]nt%.") then
    trimmed_line = trimmed_line:gsub("^(%d+%s+[Aa]nt%.)", "\\textcolor{breviaryRed}{%1}")
    trimmed_line = trimmed_line:gsub("^([Aa]nt%.)", "\\textcolor{breviaryRed}{%1}")
    trimmed_line = color_markers(trimmed_line)
    return trimmed_line .. [[ \\]]
  end

  if trimmed_line:sub(-#asterisk_marker) == asterisk_marker then
    trimmed_line = trimmed_line:sub(1, -#asterisk_marker - 1) .. asterisk_marker
    awaiting_line_after_asterisk = true
    
  elseif trimmed_line:sub(-#cross_marker) == cross_marker then
    trimmed_line = trimmed_line:sub(1, -#cross_marker - 1) .. cross_marker
    
  else
    if awaiting_line_after_asterisk then
      indentation = not indentation
      awaiting_line_after_asterisk = false
    end
  end

  trimmed_line = color_markers(trimmed_line)
  if indent_this_line then
    trimmed_line = "\\hspace*{2em}" .. trimmed_line
  end
  return trimmed_line .. [[ \\]]
end

function process_hymn_line(line)
  local trimmed_line = line:gsub("%s+$", "")

  if trimmed_line == "" then
    return "%"
  end

  if trimmed_line:match("^\\end{hymn}") or trimmed_line:sub(1, 6) == "\\input" then
    return line
  end

  local number, verse = trimmed_line:match("^(%d+)%s+(.*)$")
  if number then
    return "\\noindent\\makebox[1.2em][l]{\\textcolor{breviaryRed}{\\textbf{"
      .. number .. "}}} " .. verse .. [[ \\]]
  end

  return "\\noindent\\hspace*{1.2em}" .. trimmed_line .. [[ \\]]
end

local reading_started = false
local reading_mode = nil
local responsory_started = false
local responsory_mode = nil

function reset_reading_state()
  reading_started = false
  reading_mode = nil
end

function process_reading_line(line)
  local trimmed_line = line:gsub("^%s+", ""):gsub("%s+$", "")
  if trimmed_line == "" then
    if reading_mode ~= "plain" then
      return line
    end
    return "\\par\\smallskip%"
  end
  if trimmed_line:match("^\\input") then
    return line
  end
  if reading_mode == nil then
    reading_mode = trimmed_line:sub(1, 1) == "\\" and "tex" or "plain"
  end
  if reading_mode == "tex" then
    if trimmed_line:sub(1, 1) == "\\" then
      reading_started = true
    end
    return line
  end
  if trimmed_line:sub(1, 1) == "\\" then
    return line
  end
  if not reading_started then
    reading_started = true
    return "\\sectionHeading{CZYTANIE KRÓTKIE}{" .. trimmed_line .. "}"
  end
  return "\\noindent " .. trimmed_line .. "\\par"
end

function reset_responsory_state()
  responsory_started = false
  responsory_mode = nil
end

function process_responsory_line(line)
  local trimmed_line = line:gsub("^%s+", ""):gsub("%s+$", "")
  if trimmed_line == "" then
    if responsory_mode ~= "plain" then
      return line
    end
    return "\\par\\smallskip%"
  end
  if trimmed_line:match("^\\input") then
    return line
  end
  if responsory_mode == nil then
    responsory_mode = trimmed_line:sub(1, 1) == "\\" and "tex" or "plain"
  end
  if responsory_mode == "tex" then
    if trimmed_line:sub(1, 1) == "\\" then
      responsory_started = true
    end
    return line
  end
  if trimmed_line:sub(1, 1) == "\\" then
    return line
  end
  local output = ""
  if not responsory_started then
    responsory_started = true
    output = "\\sectionTitle{RESPONSORIUM}"
  end
  local speaker, text = trimmed_line:match("^(K%.)%s*(.*)$")
  if not speaker then
    speaker, text = trimmed_line:match("^(W%.)%s*(.*)$")
  end
  if speaker then
    text = color_markers(text)
    return output .. "\\responsoryLine{" .. speaker .. "}{" .. text .. "}"
  end
  return output .. "\\noindent " .. color_markers(trimmed_line) .. "\\par"
end

local canticle_antiphon = nil

function reset_canticle_antiphon()
  canticle_antiphon = nil
end

function process_canticle_antiphon_line(line)
  local trimmed_line = line:gsub("^%s+", ""):gsub("%s+$", "")
  if trimmed_line == "" then
    return "%"
  end
  if trimmed_line:sub(1, 1) == "\\" then
    return line
  end

  local antiphon = trimmed_line:match("^Ant%.%s*(.*)$")
  if antiphon then
    if canticle_antiphon ~= nil then
      error("Plik antyfony pieśni Maryi może zawierać tylko jedną antyfonę")
    end
    canticle_antiphon = antiphon
  elseif canticle_antiphon ~= nil then
    canticle_antiphon = canticle_antiphon .. " " .. trimmed_line
  else
    error("Antyfona pieśni Maryi musi zaczynać się od 'Ant.'")
  end

  return "\\setCanticleAntiphon{" .. color_markers(canticle_antiphon) .. "}"
end

function validate_canticle_antiphon()
  if canticle_antiphon == nil or canticle_antiphon == "" then
    error("Brak antyfony pieśni Maryi; oczekiwano tekstu zaczynającego się od 'Ant.'")
  end
end

function process_prayer_line(line)
  local trimmed_line = line:gsub("^%s+", ""):gsub("%s+$", "")
  if trimmed_line == "" then
    return "\\par\\smallskip%"
  end
  if trimmed_line:match("^\\input") or trimmed_line:match("^\\end{prosby}") then
    return line
  end
  if trimmed_line == "Można użyć krótszej formy próśb." then
    return "\\noindent\\textcolor{breviaryRed}{Można użyć }\\textcolor{breviaryGreen}{\\textbf{krótszej formy próśb.}}\\par\\medskip"
  end
  if trimmed_line:match("^%-") then
    local request = trimmed_line:gsub("^%-%s*", "")
    return "\\noindent\\textcolor{breviaryRed}{-} " .. request .. "\\par"
  end
  if trimmed_line == "Zachowaj lud swój, o Panie!" then
    return "\\noindent\\textit{" .. trimmed_line .. "}\\par"
  end
  return "\\noindent " .. trimmed_line .. "\\par"
end

function process_final_prayer_line(line)
  local trimmed_line = line:gsub("^%s+", ""):gsub("%s+$", "")
  if trimmed_line == "" then
    return "\\par\\smallskip%"
  end
  if trimmed_line:match("^\\input") or trimmed_line:match("^\\end{modlitwa_koncowa}") then
    return line
  end
  return "\\noindent " .. color_markers(trimmed_line) .. "\\par"
end
