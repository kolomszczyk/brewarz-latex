% Definicja koloru MUSI być w LaTeX-ie (przed luacode)
\definecolor{brewiarzCzerwony}{HTML}{B40000}


\begin{luacode*}
function obsluguj_brewiarz(linia)
  local spakowana = linia:gsub("%s+$", "")
  
  if spakowana == "" then
    return "%"
  end

  if spakowana:match("^\\end{brewiarz}") then
    return linia
  end

  if wciecie then
    spakowana = "\\hspace*{2em}" .. spakowana
  end

  if spakowana:find("%*$") then
    spakowana = spakowana:gsub("%*$", "\\textcolor{brewiarzCzerwony}{*}")
    oczekuje_linii_po_gwiazdce = true
    
  elseif spakowana:find("†$") then
    spakowana = spakowana:gsub("†$", "\\textcolor{brewiarzCzerwony}{†}")
    
  else
    if oczekuje_linii_po_gwiazdce then
      wciecie = not wciecie
      oczekuje_linii_po_gwiazdce = false
    end
  end

  return spakowana .. [[ \\]]
end
\end{luacode*}

\newenvironment{brewiarz}{%
  \par
  \setlength{\parindent}{0pt}%
  \directlua{wciecie= false}
  \directlua{oczekuje_linii_po_gwiazdce = false}
  \directlua{luatexbase.add_to_callback("process_input_buffer", obsluguj_brewiarz, "brewiarz_filter")}
}{%
  \directlua{luatexbase.remove_from_callback("process_input_buffer", "brewiarz_filter")}
  \par
}
