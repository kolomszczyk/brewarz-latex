\begin{luacode*}
function obsluguj_brewiarz(linia)
  local spakowana = linia:gsub("%s+$", "")
  
  if spakowana == "" then
    return "%"
  end

  if spakowana:match("^\\end{brewiarz}") then
    return linia
  end

  if wciecie then
    spakowana = "\\hspace*{2em}" .. spakowana
  end

  if spakowana:find("%*$") then
    spakowana = spakowana:gsub("%*$", "\\textcolor{brewiarzCzerwony}{*}")
    oczekuje_linii_po_gwiazdce = true
    
  elseif spakowana:find("†$") then
    spakowana = spakowana:gsub("†$", "\\textcolor{brewiarzCzerwony}{†}")
    
  else
    if oczekuje_linii_po_gwiazdce then
      wciecie = not wciecie
      oczekuje_linii_po_gwiazdce = false
    end
  end

  return spakowana .. [[ \\]]
end
function process_hymn_line(line)
  local trimmed_line = line:gsub("%s+$", "")

  if trimmed_line == "" then
    return "\\par\\vspace{0.1em}%"
  end

  if trimmed_line:match("^\\end{hymn}") or trimmed_line:sub(1, 6) == "\\input" then
    return line
  end

  local number, verse = trimmed_line:match("^(%d+)%s+(.*)$")
  if number then
    return "\\noindent\\makebox[1.2em][l]{\\textcolor{breviaryRed}{\\textbf{"
      .. number .. "}}} " .. verse .. [[ \\]]
  end

  return "\\noindent\\hspace*{1.2em}" .. trimmed_line .. [[ \\]]
end

\newenvironment{brewiarz}{%
  \par
  \setlength{\parindent}{0pt}%
  \directlua{wciecie= false}
  \directlua{oczekuje_linii_po_gwiazdce = false}
  \directlua{luatexbase.add_to_callback("process_input_buffer", obsluguj_brewiarz, "brewiarz_filter")}
}{%
  \directlua{luatexbase.remove_from_callback("process_input_buffer", "brewiarz_filter")}
  \par
}

\newenvironment{hymn}{%
  \clearpage
  \par
  \setlength{\parindent}{0pt}%
  \directlua{luatexbase.add_to_callback("process_input_buffer", process_hymn_line, "hymn_filter")}
}{%
  \directlua{luatexbase.remove_from_callback("process_input_buffer", "hymn_filter")}
  \par
  \clearpage
}
