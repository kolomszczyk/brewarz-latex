\begin{luacode*}
local asterisk_marker = "*"
local cross_marker = "†"
local block_number = 1
local indentation = false
local awaiting_line_after_as erisk = false

local function color_markers(line)
  line = line:gsub("%*", "\\textcolor{breviaryRed}{*}")
  line = line:gsub("†", "\\textcolor{breviaryRed}{†}")
  line = line:gsub("/", "\\textcolor{breviaryRed}{/}")
  return line
end

function reset_psalmody_state()
  block_number = 1
  indentation = false
  awaiting_line_after_asterisk = false
end

function process_breviary_line(line)
  local trimmed_line = line:gsub("%s+$", "")
  
  if trimmed_line == "" then
    block_number = block_number + 1
    indentation = false
    awaiting_line_after_asterisk = false
    return "\\par\\vspace{\\psalmodyblockskip}%"
  end

  if trimmed_line:match("^\\end{psalmody}") then
    return line
  end

  if trimmed_line:sub(1, 6) == "\\input" then
    return line
  end

  local indent_this_line = indentation

  if block_number == 2 then
    return "\\noindent\\makebox[\\linewidth][c]{\\textcolor{breviaryRed}{\\textbf{"
      .. trimmed_line .. "}}}\\\\"
  end

  if trimmed_line:match("^%d+%s+[Aa]nt%.") or trimmed_line:match("^[Aa]nt%.") then
    trimmed_line = trimmed_line:gsub("^(%d+%s+[Aa]nt%.)", "\\textcolor{breviaryRed}{%1}")
    trimmed_line = trimmed_line:gsub("^([Aa]nt%.)", "\\textcolor{breviaryRed}{%1}")
    trimmed_line = color_markers(trimmed_line)
    return trimmed_line .. [[ \\]]
  end

  if trimmed_line:sub(-#asterisk_marker) == asterisk_marker then
    trimmed_line = trimmed_line:sub(1, -#asterisk_marker - 1) .. asterisk_marker
    awaiting_line_after_asterisk = true
    
  elseif trimmed_line:sub(-#cross_marker) == cross_marker then
    trimmed_line = trimmed_line:sub(1, -#cross_marker - 1) .. cross_marker
    
  else
    if awaiting_line_after_asterisk then
      indentation = not indentation
      awaiting_line_after_asterisk = false
    end
  end

  trimmed_line = color_markers(trimmed_line)
  if indent_this_line then
    trimmed_line = "\\hspace*{2em}" .. trimmed_line
  end
  return trimmed_line .. [[ \\]]
end

function process_hymn_line(line)
  local trimmed_line = line:gsub("%s+$", "")

  if trimmed_line == "" then
    return "\\par\\vspace{0.1em}%"
  end

  if trimmed_line:match("^\\end{hymn}") or trimmed_line:sub(1, 6) == "\\input" then
    return line
  end

  local number, verse = trimmed_line:match("^(%d+)%s+(.*)$")
  if number then
    return "\\noindent\\makebox[1.2em][l]{\\textcolor{breviaryRed}{\\textbf{"
      .. number .. "}}} " .. verse .. [[ \\]]
  end

  return "\\noindent\\hspace*{1.2em}" .. trimmed_line .. [[ \\]]
end
\end{luacode*}

\newlength{\psalmodyblockskip}
\setlength{\psalmodyblockskip}{1em}
\newcounter{psalmodysection}

\newenvironment{psalmody}{%
  \par
  \setlength{\parindent}{0pt}%
  \stepcounter{psalmodysection}%
  \sectionTitle{PSALMODIA \Roman{psalmodysection}}%
  \directlua{reset_psalmody_state()}
  \directlua{luatexbase.add_to_callback("process_input_buffer", process_breviary_line, "psalmody_filter")}
}{%
  \directlua{luatexbase.remove_from_callback("process_input_buffer", "psalmody_filter")}
  \par
}

\newenvironment{hymn}{%
  \clearpage
  \par
  \setlength{\parindent}{0pt}%
  \sectionTitle{HYMN}%
  \directlua{luatexbase.add_to_callback("process_input_buffer", process_hymn_line, "hymn_filter")}
}{%
  \directlua{luatexbase.remove_from_callback("process_input_buffer", "hymn_filter")}
  \par
  \clearpage
}
